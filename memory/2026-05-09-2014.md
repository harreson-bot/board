### **Authentication: JWT Format (NEW)**

Each request needs a freshly-generated JWT with:
```json
{
  "iss": "cdp",
  "sub": "organizations/{org_id}/apiKeys/{key_id}",
  "nbf": <current_timestamp>,
  "exp": <current_timestamp + 120>,
  "uri": "GET api.coinbase.com/api/v3/brokerage/accounts"
}
```

Headers include:
```json
{
  "alg": "ES256",
  "typ": "JWT",
  "kid": "organizations/{org_id}/apiKeys/{key_id}",
  "nonce": "<random_hex_32_chars>"
}
```

### **JavaScript Implementation (Correct)**

```javascript
const jwt = require('jsonwebtoken');
const crypto = require('crypto');

const API_KEY = 'organi…_id}';
const PRIVATE_KEY = '***\n';

function buildJWT(method, path) {
  const uri = `${method} api.coinbase.com${path}`;
  const nonce = crypto.randomBytes(16).toString('hex');
  
  return jwt.sign(
    {
      iss: 'cdp',
      sub: API_KEY,
      nbf: Math.floor(Date.now() / 1000),
      exp: Math.floor(Date.now() / 1000) + 120,
      uri: uri,
    },
    PRIVATE_KEY,
    {
      algorithm: 'ES256',
      header: {
        kid: API_KEY,
        nonce: nonce,
      },
    }
  );
}

// Use it:
const jwt_token = buildJWT('GET', '/api/v3/brokerage/accounts');
// Then: curl -H "Authorization: Bearer ${jwt_token}" https://api.coinbase.com/api/v3/brokerage/accounts
```
